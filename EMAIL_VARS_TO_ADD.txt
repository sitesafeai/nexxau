FROM_NAME=SiteSafe

# ============================================
# SETUP INSTRUCTIONS:
# ============================================
#
# OPTION 1: Gmail (Easiest for Testing)
# --------------------------------------
# 1. Enable 2-Factor Authentication on your Google account
# 2. Visit: https://myaccount.google.com/apppasswords
# 3. Select App: "Mail", Device: "Other (Custom name)"
# 4. Name it "SiteSafe"  
# 5. Copy the 16-character password (format: xxxx xxxx xxxx xxxx)
# 6. Replace "your-email@gmail.com" with your Gmail address
# 7. Replace "xxxx xxxx xxxx xxxx" with the App Password
# 8. Save the .env file
# 9. Run: cd /Users/luizcarneiro/nexxau/app && node test-email.js
#
# OPTION 2: Other SMTP Provider
# ------------------------------
# Update SMTP_HOST, SMTP_PORT according to your provider:
#
# SendGrid:
#   SMTP_HOST=smtp.sendgrid.net
#   SMTP_PORT=587
#   SMTP_USER=apikey
#   SMTP_PASSWORD=your-sendgrid-api-key
#
# AWS SES:
#   SMTP_HOST=email-smtp.us-east-1.amazonaws.com
#   SMTP_PORT=587
#   SMTP_USER=your-ses-smtp-username
#   SMTP_PASSWORD=your-ses-smtp-password
#
# ============================================
